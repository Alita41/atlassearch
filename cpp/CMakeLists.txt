cmake_minimum_required(VERSION 3.16)
project(atlassearch_bm25 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(atlassearch_core src/search_engine.cpp)
target_include_directories(atlassearch_core PUBLIC include)
add_executable(atlassearch_bm25 src/main.cpp)
target_link_libraries(atlassearch_bm25 PRIVATE atlassearch_core)
add_executable(atlassearch_cpp_tests tests/test_search.cpp)
target_link_libraries(atlassearch_cpp_tests PRIVATE atlassearch_core)
enable_testing()
add_test(NAME atlassearch_cpp_tests COMMAND atlassearch_cpp_tests)
